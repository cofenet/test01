const Store = {
    _storageKey: 'local_github_reader_data',

    // 1. 浠庢湰鍦扮紦瀛樿鍙栫姸鎬?
    _loadFromStorage() {
        try {
            const raw = localStorage.getItem(this._storageKey);
            if (raw) {
                const data = JSON.parse(raw);
                return {
                    history: data.history || [],
                    readIds: data.readIds || [],
                    starIds: data.starIds || []
                };
            }
        } catch (e) {
            console.warn('璇诲彇鏈湴缂撳瓨澶辫触锛屽皢浣跨敤榛樿鐘舵€?, e);
        }
        return { history: [], readIds: [], starIds: [] };
    },

    // 2. 灏嗙姸鎬佸啓鍏ユ湰鍦扮紦瀛?
    _saveToStorage() {
        try {
            const dataToSave = {
                history: this.state.history,
                readIds: this.state.readIds,
                starIds: this.state.starIds
            };
            localStorage.setItem(this._storageKey, JSON.stringify(dataToSave));
        } catch (e) {
            console.error('鍐欏叆鏈湴缂撳瓨澶辫触', e);
        }
    },

    // 3. 鍒濆鍖栫姸鎬侊紙鍦ㄨ繖閲岀洿鎺ヨ皟鐢ㄦ柟娉曪紝鑰屼笉鏄敤灞曞紑杩愮畻绗﹁嚜寮曠敤锛?
    state: {
        newsList: [],
        history: [],
        readIds: [],
        starIds: []
    },

    // 4. 鍦?Store 瀹氫箟鍚庯紝绔嬪嵆鐢ㄧ紦瀛樻暟鎹鐩栧垵濮嬬姸鎬?
    init() {
        const cached = this._loadFromStorage();
        this.state.history = cached.history;
        this.state.readIds = cached.readIds;
        this.state.starIds = cached.starIds;
    },

    // 5. 鏂囩珷鍒楄〃鐩稿叧锛堜粎鍐呭瓨鎿嶄綔锛?
    getArticles() { return this.state.newsList; },
    setArticles(list) { this.state.newsList = list || []; },

    // 6. 鍘嗗彶璁板綍鐩稿叧
    getHistory() { return this.state.history; },
    addHistory(item) {
        this.state.history = [item, ...this.state.history.filter(h => h.id !== item.id)].slice(0, 20);
        this._saveToStorage();
    },
    clearHistory() { 
        this.state.history = []; 
        this._saveToStorage();
    },

    // 7. 宸茶鐘舵€佺浉鍏?
    isRead(id) { return this.state.readIds.includes(id); },
    toggleRead(id) {
        const idx = this.state.readIds.indexOf(id);
        idx > -1 ? this.state.readIds.splice(idx, 1) : this.state.readIds.push(id);
        this._saveToStorage();
    },
    markAllRead(ids) { 
        this.state.readIds = [...new Set([...this.state.readIds, ...ids])]; 
        this._saveToStorage();
    },
    clearRead() { 
        this.state.readIds = []; 
        this._saveToStorage();
    },

    // 8. 鏀惰棌鐘舵€佺浉鍏?
    isStar(id) { return this.state.starIds.includes(id); },
    toggleStar(id) {
        const idx = this.state.starIds.indexOf(id);
        idx > -1 ? this.state.starIds.splice(idx, 1) : this.state.starIds.push(id);
        this._saveToStorage();
    },
    clearStar() { 
        this.state.starIds = []; 
        this._saveToStorage();
    },

    // 9. 閫氱敤鐘舵€佺鐞?
    setState(key, value) { this.state[key] = value; },
    getState(key) { return this.state[key]; }
};

const Render = {
    renderList(articles) {

            box.innerHTML = '<div style="padding:24px;text-align:center;color:var(--text-secondary)">鏆傛棤鍐呭</div>';
            return;
        }
        box.innerHTML = articles.map(a => {
            const read = Store.isRead(a.id) ? ' read' : '';
            const star = Store.isStar(a.id) ? '猸? : '鈽?;
            return `<div class="list-item${read}" data-id="${a.id}" role="listitem">
                <div class="item-title">${safeHTML(a.title)}</div>
                <div class="item-meta">
                    <span>${safeHTML(a.source || '')}</span>

                    <span class="star-icon" data-star="${a.id}">${star}</span>

     * - 棣栨鐐瑰嚮锛氬厛娓叉煋鍩烘湰淇℃伅锛堢畝浠?鏄熸爣/璇█锛夛紝鍐嶅紓姝ユ媺鍙?README 杩藉姞鍒颁笅鏂?     * - 鍐嶆鐐瑰嚮锛氱洿鎺ヤ娇鐢ㄧ紦瀛橈紝绉掑紑涓斾笉閲嶅娑堣€?API 閰嶉
     * - 鍘嗗彶璁板綍鐐瑰嚮杩涘叆鐨勬枃绔犲悓鏍疯蛋杩欏娴佺▼
     * @param {Object} article - 鏂囩珷瀵硅薄锛坱itle 鍗充粨搴撳叏鍚嶏紝濡?roboflow/sports锛?     * @param {HTMLElement|null} itemEl - 瑙﹀彂鐐瑰嚮鐨勫垪琛ㄩ」鑺傜偣
     */
    async openArticle(article, itemEl) {
        if (!article) return;

        // 1. 鏍囪宸茶锛屽苟鍚屾鍒楄〃涓殑鏈灏忓渾鐐?        if (!Store.isRead(article.id)) {
            Store.toggleRead(article.id);
            // 鈥滄湭璇烩€濊鍥句笅锛岀偣寮€鍗冲凡璇伙紝搴旂洿鎺ヤ粠鍒楄〃绉婚櫎
            if (this.currentCat === 'unread') {
                this.currentFiltered = this.currentFiltered.filter(a => a.id !== article.id);
            }
            Render.renderList(this.currentFiltered);
            Render.updateStat(this.currentFiltered);
        }

        // 2. 楂樹寒閫変腑椤癸紙鍒楄〃鍙兘鍒氳閲嶆覆鏌擄紝闇€閲嶆柊瀹氫綅鑺傜偣锛?        document.querySelectorAll('.list-item').forEach(i => i.classList.remove('active'));
        const freshItem = document.querySelector(`.list-item[data-id="${article.id}"]`) || itemEl;
        if (freshItem) freshItem.classList.add('active');

        // 3. README 宸茬紦瀛樿繃锛氱洿鎺ユ覆鏌擄紙绗簩娆＄偣鍑荤寮€锛屼笉鍐嶈姹傜綉缁滐級
        if (article._readmeLoaded) {
            Render.renderDetail(article);
            return;
        }

        // 4. 鍏堟覆鏌撳熀鏈俊鎭紙绠€浠?/ 鏄熸爣 / 璇█锛?        Render.renderDetail(article);

        // 闃叉蹇€熷弻鍑婚€犳垚閲嶅璇锋眰
        if (article._readmeLoading) return;
        article._readmeLoading = true;
        this._openingId = article.id;

        // 5. 鍦ㄨ鎯呭尯搴曢儴杩藉姞鈥滃姞杞戒腑鈥濆崰浣?        const detailBox = document.getElementById('detailBox');
        document.getElementById('readmeLoading')?.remove();
        if (detailBox) {
            const loadingEl = document.createElement('div');
            loadingEl.id = 'readmeLoading';
            loadingEl.style.cssText = 'padding:24px;text-align:center;opacity:0.7;';
            loadingEl.textContent = '\u23F3 姝ｅ湪鍔犺浇椤圭洰 README 鏂囨。...';
            detailBox.appendChild(loadingEl);
        }

        // 6. 鎳掑姞杞?README锛坅rticle.title 灏辨槸浠撳簱鍏ㄥ悕锛屽唴閮ㄥ甫缂撳瓨涓庡绾у閿欙級
        const result = await API.fetchReadme(article.title);

        article._readmeLoading = false;
        // 濡傛灉璇锋眰鏈熼棿鐢ㄦ埛宸插垏鎹㈠埌鍏朵粬鏂囩珷锛屼涪寮冭繃鏈熺粨鏋滐紝闃叉鍐呭涓叉壈
        if (this._openingId !== article.id) return;

        document.getElementById('readmeLoading')?.remove();

        let readmeSection;
        if (result && result.ok) {
            readmeSection = '<hr style="border:0;border-top:1px solid var(--border-color,#30363d);margin:24px 0;">' +
                '<div class="markdown-body">' + result.html + '</div>';
            // 7. 浠呭湪鎴愬姛鏃剁紦瀛樺埌鏂囩珷瀵硅薄锛氳拷鍔犺繘 content锛屼箣鍚庝粠鍘嗗彶璁板綍鐐瑰紑涔熻兘鐩存帴鐪嬪埌鍏ㄦ枃
            article._readmeLoaded = true;
            article.content = (article.content || '') + readmeSection;
        } else {
            // 澶辫触锛氭樉绀哄叿浣撳師鍥狅紝涓斾笉鍐欏叆 content銆佷笉鏍囪 _readmeLoaded 鈥斺€?鍐嶆鐐瑰嚮鏈」鐩嵆鍙嚜鍔ㄩ噸璇?            const reason = (result && result.message) ? result.message : 'README 鍔犺浇澶辫触锛岀偣鍑绘湰椤圭洰鍙噸璇曘€?;
            readmeSection = '<p style="opacity:0.75;padding:16px 0;">馃搫 ' + reason + '</p>';
            console.warn('鉂?[APP] README 鍔犺浇澶辫触:', result);
        }

        // 8. 杩藉姞鍒板綋鍓嶈鎯呭尯锛堜笉鏁翠綋閲嶆覆鏌擄紝閬垮厤婊氬姩浣嶇疆璺冲洖椤堕儴锛?        const box = document.getElementById('detailBox');
        if (box) {
            const wrap = document.createElement('div');
            wrap.innerHTML = readmeSection;
            while (wrap.firstChild) box.appendChild(wrap.firstChild);
        }
    },

    bindEvents() {
        // 鍒嗙被鍒囨崲
        document.querySelectorAll('.tab-btn').forEach(btn => {
            btn.addEventListener('click', async () => {
                document.querySelectorAll('.tab-btn').forEach(b => {
                    b.classList.remove('active');
                    b.setAttribute('aria-selected','false');
                });
                btn.classList.add('active');
                btn.setAttribute('aria-selected','true');
                this.currentCat = btn.dataset.cat;
                // 鍒囨崲鍒嗙被鏃惰Е鍙戝紓姝ユ暟鎹媺鍙?                await this.filterArticles();
            });
        });

        // 鎼滅储锛堢函鏈湴杩囨护锛屾棤闇€閲嶆柊璇锋眰缃戠粶锛?        const searchInput = document.getElementById('searchInput');
        if (searchInput) searchInput.addEventListener('input', debounce(() => this.filterArticles()));

        // 鍒楄〃鐐瑰嚮锛堜簨浠跺鎵橈級
        document.getElementById('listBox')?.addEventListener('click', e => {
            const starEl = e.target.closest('[data-star]');
            if (starEl) {
                Store.toggleStar(starEl.dataset.star);
                this.filterArticles();
                showToast(Store.isStar(starEl.dataset.star) ? '宸叉敹钘? : '宸插彇娑堟敹钘?);
                return;
            }
            const item = e.target.closest('.list-item');
            if (item) {
                const article = this.currentFiltered.find(a => a.id === item.dataset.id);
                // 鏀逛负 openArticle锛氬厛娓叉煋鍩烘湰淇℃伅锛屽啀鎳掑姞杞?README 鍏ㄦ枃
                if (article) this.openArticle(article, item);
            }
        });

        // 鍘嗗彶璁板綍鐐瑰嚮
        document.getElementById('historyItems')?.addEventListener('click', e => {
            const item = e.target.closest('[data-hid]');
            if (item) {
                const article = Store.getArticles().find(a => a.id === item.dataset.hid);
                // 鍘嗗彶璁板綍鍚屾牱璧版噿鍔犺浇娴佺▼锛涘凡鍔犺浇杩囩殑鏂囩珷浼氱寮€
                if (article) this.openArticle(article, null);
            }
        });

        // 宸ュ叿鏍忥細鍒锋柊鎸夐挳
        document.getElementById('refreshBtn')?.addEventListener('click', async () => {
            await this.filterArticles(); // 鐩存帴澶嶇敤 filterArticles 閫昏緫
            showToast('宸插埛鏂版渶鏂板紑婧愰」鐩?);
        });

        document.getElementById('markAllRead')?.addEventListener('click', () => {
            Store.markAllRead(this.currentFiltered.map(a=>a.id));
            this.filterArticles();
            showToast('宸插叏閮ㄦ爣涓哄凡璇?);
        });
        document.getElementById('clearAllRead')?.addEventListener('click', () => {
            Store.clearRead();
            this.filterArticles();
            showToast('宸叉竻闄ゅ凡璇绘爣璁?);
        });
        document.getElementById('clearAllStar')?.addEventListener('click', () => {
            Store.clearStar();
            this.filterArticles();
            showToast('宸叉竻绌烘敹钘?);
        });
        document.getElementById('clearHistoryAll')?.addEventListener('click', () => {
            Store.clearHistory();
            Render.renderHistory();
            showToast('宸叉竻绌哄巻鍙?);
        });

        // 瀛楀彿璋冭妭
        document.querySelectorAll('.font-btn').forEach(btn => {
            btn.addEventListener('click', () => {
                document.querySelectorAll('.font-btn').forEach(b => b.classList.remove('active'));
                btn.classList.add('active');
                document.documentElement.style.setProperty('--font-size-base', btn.dataset.root + 'px');
            });
        });

        // 涓婚鍒囨崲
        document.getElementById('themeBtn')?.addEventListener('click', () => {
            const isDark = document.documentElement.getAttribute('data-theme') === 'dark';
            document.documentElement.setAttribute('data-theme', isDark ? 'light' : 'dark');
            document.getElementById('themeBtn').textContent = isDark ? '鍒囨崲鏆楅粦妯″紡' : '鍒囨崲鏄庝寒妯″紡';
            localStorage.setItem('reader_theme', isDark ? 'light' : 'dark');
        });

        // 闃呰杩涘害鏉?        const detailBox = document.getElementById('detailBox');
        detailBox?.addEventListener('scroll', () => {
            const bar = document.getElementById('progressBar');
            if (!bar) return;
            const pct = detailBox.scrollTop / (detailBox.scrollHeight - detailBox.clientHeight) * 100;
            bar.style.width = Math.min(100, Math.max(0, pct)) + '%';
        });

        // 鍥炲埌椤堕儴
        const backTop = document.getElementById('backTop');
        const listBackTop = document.getElementById('listBackTop');
        window.addEventListener('scroll', () => {
            if (backTop) backTop.style.display = window.scrollY > 300 ? 'flex' : 'none';
        });
        backTop?.addEventListener('click', () => window.scrollTo({ top: 0, behavior: 'smooth' }));

        const listWrap = document.getElementById('listScrollWrap');
        listWrap?.addEventListener('scroll', () => {
            if (listBackTop) listBackTop.style.display = listWrap.scrollTop > 200 ? 'flex' : 'none';
        });
        listBackTop?.addEventListener('click', () => listWrap?.scrollTo({ top: 0, behavior: 'smooth' }));

        // 涓撴敞妯″紡
        document.getElementById('focusModeBtn')?.addEventListener('click', () => {
            const sidebar = document.querySelector('.left-sidebar');
            if (sidebar) {
                const hidden = sidebar.style.display === 'none';
                sidebar.style.display = hidden ? '' : 'none';
                showToast(hidden ? '閫€鍑轰笓娉ㄦā寮? : '宸茶繘鍏ヤ笓娉ㄩ槄璇绘ā寮?);
            }
        });
    },

    initTheme() {
        const saved = localStorage.getItem('reader_theme');
        if (saved === 'dark') {
            document.documentElement.setAttribute('data-theme', 'dark');
            const btn = document.getElementById('themeBtn');
            if (btn) btn.textContent = '鍒囨崲鏄庝寒妯″紡';
        }
    }
};
